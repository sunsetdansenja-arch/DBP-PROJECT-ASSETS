const actionBranch = $node["Webhook"].json.body.branch_name; let staffs = []; try { staffs = $items("GET STAFF"); } catch(e) {} // Filter Capster sesuai cabang staffs = staffs.filter(s => s.json.dbp_branch === actionBranch); let bookings = []; try { // Ambil SEMUA bookingan biar kita tahu mana yang udah keisi bookings = $items("GET BOOKING"); } catch(e) {} let packages = []; try { packages = $items("GET PACKAGES1"); } catch(e) {} const perfMap = {}; const bookedSlotsMap = {}; // Ini otak baru buat nyatet jam yang udah laku bookings.forEach(b => { const id = b.json.capster_id; const status = b.json.status; const tanggal = b.json.tanggal; // Format dari frontend: YYYY-MM-DD const jam = b.json.jam; // Format dari frontend: HH:MM // Hitung performa capster (hanya yang statusnya DONE) if (status === 'DONE') { perfMap[id] = (perfMap[id] || 0) + 1; } // BLOKIR JADWAL: Catat jam yang udah dibooking (kecuali kalau udah CANCELED) if (status !== 'CANCELED' && status !== 'REJECTED' && tanggal && jam) { if (!bookedSlotsMap[id]) bookedSlotsMap[id] = []; // Gabungin tanggal dan jam jadi format: YYYY-MM-DD_HH:MM bookedSlotsMap[id].push(`${tanggal}_${jam}`); } }); const topStaffId = Object.keys(perfMap).reduce((a, b) => perfMap[a] > perfMap[b] ? a : b, null); // Bangun data akhir siap saji ke Frontend return staffs.map(s => { const role = s.json.role || ""; const staffId = s.json.id; let modifier = 0; // Logika harga Dinamis if (role.includes("SR")) modifier = 50000; if (role.includes("MR")) modifier = 150000; const servicesWithPrices = packages.map(p => ({ id: p.json.id, name: p.json.NAME, price: Number(p.json.PRICE) + modifier })); return { json: { ...s.json, is_top_performance: staffId == topStaffId, total_done: perfMap[staffId] || 0, services: servicesWithPrices, booked_slots: bookedSlotsMap[staffId] || [] // <--- INI KUNCI BIAR JAM DI FRONTEND MATI } }; });
